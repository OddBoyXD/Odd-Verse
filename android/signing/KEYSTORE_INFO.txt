HOW TO SIGN ANY APK WITH THIS KEY (3 Easy Methods):
--------------------------------------------------------------------------------

[Method 1: Using apksigner (Recommended for CLI / Scripts)]
apksigner sign \
  --v1-signing-enabled true \
  --v2-signing-enabled true \
  --v3-signing-enabled true \
  --ks master_release.keystore \
  --ks-pass "pass:MasterKey@2026#Secure" \
  --ks-key-alias masterkey \
  --key-pass "pass:MasterKey@2026#Secure" \
  --out YourSignedApp.apk \
  YourAlignedApp.apk

[Method 2: Using PK8 + PEM (No password prompt)]
apksigner sign \
  --key master_release.pk8 \
  --cert master_release.x509.pem \
  --out YourSignedApp.apk \
  YourAlignedApp.apk

[Method 3: In Android Studio (build.gradle)]
android {
    signingConfigs {
        release {
            storeFile file("master_release.keystore")
            storePassword "MasterKey@2026#Secure"
            keyAlias "masterkey"
            keyPassword "MasterKey@2026#Secure"
            v1SigningEnabled true
            v2SigningEnabled true
        }
    }
    buildTypes {
        release {
            signingConfig signingConfigs.release

}
